import tkinter

class MPGCalculatorGUI:
    def __init__(self):
        self.main_window = tkinter.Tk()

        self.gallons_entry = tkinter.Entry(self.main_window, width=10)
        self.miles_entry = tkinter.Entry(self.main_window, width=10)
        self.result_var = tkinter.StringVar()

        tkinter.Label(self.main_window, text='Gallons:').pack()
        self.gallons_entry.pack()
        tkinter.Label(self.main_window, text='Miles:').pack()
        self.miles_entry.pack()
        tkinter.Label(self.main_window, textvariable=self.result_var).pack()
        tkinter.Button(self.main_window, text='Calculate MPG',
                       command=self.calculate_mpg).pack()

        tkinter.mainloop()

    def calculate_mpg(self):
        gallons = float(self.gallons_entry.get())
        miles = float(self.miles_entry.get())
        self.result_var.set(f'{miles / gallons:.2f} MPG')

MPGCalculatorGUI()
